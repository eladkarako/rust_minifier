3#[derive(Parser, Debug, Clone)]
4#[command(name = env!("CARGO_PKG_NAME"))]
5#[command(version = env!("CARGO_PKG_VERSION"))]
6#[command(author = env!("CARGO_PKG_AUTHORS"))]
7#[command(about = env!("CARGO_PKG_DESCRIPTION"), long_about = None)]
8#[command(
9    after_help = concat!("Repository:    ", env!("CARGO_PKG_REPOSITORY"), "\n", "Homepage:      ", env!("CARGO_PKG_HOMEPAGE"), "\n", "Use --list-types to see every supported file type.")
10)]
11pub struct Args {
12    /// Input files or directories to minify.
13    ///
14    /// Can be individual supported text files, or directories to recursively search.
15    /// If omitted, reads from stdin.
16    #[arg(value_name = "FILES")]
17    pub files: Vec<String>,
18
19    /// Force a language for every input instead of detecting it.
20    ///
21    /// Accepts a language id or alias from `--list-types`, for example
22    /// `html`, `typescript`, `kotlin` or `yaml`.
23    #[arg(short = 't', long = "type", value_name = "LANGUAGE")]
24    pub forced_type: Option<String>,
25
26    /// Keep original line breaks where the format allows it.
27    ///
28    /// By default output is collapsed to a single line. Line based formats
29    /// already keep their line structure.
30    #[arg(long)]
31    pub preserve_line_breaks: bool,
32
33    /// List every supported and excluded file type, then exit.
34    #[arg(long)]
35    pub list_types: bool,
36
37    /// Suffix to append to minified output filenames.
38    ///
39    /// For example, with suffix `_minified`, `main.rs` becomes `main_minified.rs`.
40    /// Only used when writing individual files (not with `--concat-to`).
41    #[arg(short, long, default_value = "_minified")]
42    pub suffix: String,
43
44    /// Destination for concatenated output.
45    ///
46    /// Can be A file path (writes minified content to that file), a directory path (creates a timestamped file in that directory), `STDOUT` (or empty - prints to standard output, default).
47    ///
48    /// When specified, individual minified files are not written to disk.
49    #[arg(
50        long,
51        value_name = "DESTINATION",
52        default_value = "STDOUT"
53    )]
54    pub concat_to: String,
55
56    /// Include source file paths as comments in output.
57    ///
58    /// When enabled, each minified file is prefixed with a comment containing its full path.
59    /// Useful for tracking which source files contributed to concatenated output.
60    #[arg(long, default_value = "true")]
61    pub include_file_path: bool,
62
63    /// Do not prefix output with the source file path.
64    ///
65    /// This is the inverse of `--include-file-path` and wins when both are set.
66    #[arg(long)]
67    pub no_include_file_path: bool,
68
69    /// Exclude files matching these patterns (non-case-sensitive substring match on filename).
70    ///
71    /// Files whose names contain any of these patterns are skipped during processing.
72    /// Matching is case-insensitive and applied only to the filename, not the full path.
73    ///
74    /// for example:
75    /// rust_minifier src/ --exclude test --exclude mock
76    /// will exclude files like: `test_helper.rs`, `mock_data.toml`, `my_test.rs`
77    #[arg(long)]
78    pub exclude: Vec<String>,
79
80    /// Try to limit maximum number of characters per file in --concat-to result, by generating multiple files each with a '_part_XX_of_YY' suffix.
81    #[arg(long, default_value = "unlimited")]
82    pub soft_split: String,
83}
84
85impl Args {
86    pub fn parse() -> Args {

88    }
89
90    /// Whether source paths should be embedded in the output.
91    pub fn include_source_path(&self) -> bool {
92        self.include_file_path && !self.no_include_file_path
93    }
94
95    /// Build the shared minifier options from the CLI flags.
96    pub fn minifier_options(
97        &self

99        crate::minifier::MinifierOptions {
100            preserve_line_breaks: self.preserve_line_breaks,

102    }
103}
